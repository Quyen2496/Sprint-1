Mã Nguồn Hàm API Backend Độc Lập (JavaScript / Node.js)

Code xử lý hoàn chỉnh 5 tiêu chí: Nhận payload, Query DB, SHA-256 Hash, So sánh & Response bảo mật.

/**
 * Hàm mô phỏng API Đăng nhập Backend
 * @param {Object} req - HTTP Request object ({ body: { email, password } })
 * @returns {Promise<Object>} - HTTP Response JSON & Status Code
 */
async function loginApi(req) {
    const { email, password } = req.body || {};

    // 1. Kiểm tra đầu vào tối thiểu
    if (!email || !password) {
        return {
            status: 400,
            json: { success: false, message: "Email và mật khẩu không được để trống." }
        };
    }

    // 2. Tìm kiếm tài khoản trong Database theo Email
    const user = await mockDb.findOne({ email: email.toLowerCase() });

    // 3. Hàm tạo SHA-256 Hash cho password input
    const hashPassword = async (plainText) => {
        const encoder = new TextEncoder();
        const data = encoder.encode(plainText);
        const hashBuffer = await crypto.subtle.digest('SHA-256', data);
        return Array.from(new Uint8Array(hashBuffer))
            .map(b => b.toString(16).padStart(2, '0')).join('');
    };

    const inputPasswordHash = await hashPassword(password);

    // 4. MẬT KHẨU / EMAIL SAI -> Trả thông báo lỗi CHUNG để bảo mật
    // Tránh tiết lộ tài khoản có tồn tại trong hệ thống hay không
    if (!user || user.passwordHash !== inputPasswordHash) {
        return {
            status: 401,
            json: {
                success: false,
                code: "AUTH_FAILED",
                message: "Email hoặc mật khẩu không chính xác."
            }
        };
    }

    // 5. Đăng nhập THÀNH CÔNG -> Trả về dữ liệu người dùng & Access Token
    return {
        status: 200,
        json: {
            success: true,
            message: "Đăng nhập thành công.",
            data: {
                token: `mock_jwt_token_${Date.now()}`,
                user: {
                    id: user.id,
                    email: user.email,
                    fullName: user.fullName
                }
            }
        }
    };
}
